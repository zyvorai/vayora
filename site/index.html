Create an order, take a payment, verify the webhook.

vayora · simulated
# amounts are integer paise (₹499.00 = 49900)
curl localhost:8080/v1/orders \
  -H "Authorization: Bearer sk_test_123" \
  -H "Idempotency-Key: order-1001" \
  -H "Content-Type: application/json" \
  -d '{"amount_paise": 49900}'

curl localhost:8080/v1/payments \
  -H "Authorization: Bearer sk_test_123" \
  -H "Idempotency-Key: pay-1001" \
  -H "Content-Type: application/json" \
  -d '{"order_id": "order_…", "vpa": "asha@okvayora"}'
import { Vayora } from "@vayora/node";

const vayora = new Vayora({ apiKey: process.env.VAYORA_API_KEY! });

const order = await vayora.orders.create({
  amountPaise: 49900,
  idempotencyKey: "order-1001",
});

const payment = await vayora.payments.create({
  orderId: order.id,
  vpa: "asha@okvayora",
  idempotencyKey: "pay-1001",
});
// payment.status === "captured" | "failed"
import { verifyWebhook } from "@vayora/node";

// use the RAW body, not re-serialised JSON
app.post("/webhooks", express.raw({ type: "*/*" }), (req, res) => {
  const ok = verifyWebhook({
    secret: process.env.VAYORA_WEBHOOK_SECRET!,
    header: req.get("Vayora-Signature")!,
    rawBody: req.body,
  });
  if (!ok) return res.sendStatus(400);
  // dedupe on event.id: delivery is at-least-once
  res.sendStatus(200);
});
